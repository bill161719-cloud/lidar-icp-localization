cmake_minimum_required(VERSION 3.20)
project(lidar_icp_localization VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# The bootstrap executable intentionally needs only a C++17 toolchain.
option(LIDAR_ENABLE_PCL "Require PCL/Eigen and validate the dependency interface" OFF)

add_library(lidar_project_options INTERFACE)
target_compile_features(lidar_project_options INTERFACE cxx_std_17)
target_include_directories(lidar_project_options INTERFACE "${PROJECT_SOURCE_DIR}/include")
if(MSVC)
  target_compile_options(lidar_project_options INTERFACE /W4 /permissive- /utf-8)
else()
  target_compile_options(lidar_project_options INTERFACE -Wall -Wextra -Wpedantic)
endif()

if(LIDAR_ENABLE_PCL)
  # Some VTK/PCL packages expose MPI C targets.
  enable_language(C)
  find_package(Eigen3 3.3 REQUIRED NO_MODULE)
  find_package(PCL 1.12 REQUIRED COMPONENTS common io filters registration visualization)
  add_library(lidar_dependencies INTERFACE)
  target_link_libraries(lidar_dependencies INTERFACE Eigen3::Eigen ${PCL_LIBRARIES})
  target_include_directories(lidar_dependencies SYSTEM INTERFACE ${PCL_INCLUDE_DIRS})
  target_link_directories(lidar_dependencies INTERFACE ${PCL_LIBRARY_DIRS})
  target_compile_options(lidar_dependencies INTERFACE ${PCL_DEFINITIONS})
  add_executable(lidar_dependency_check tests/dependency_check.cpp)
  target_link_libraries(lidar_dependency_check PRIVATE lidar_project_options lidar_dependencies)
  message(STATUS "PCL ${PCL_VERSION}; Eigen ${Eigen3_VERSION}")
endif()

add_executable(lidar_icp src/main.cpp)
target_link_libraries(lidar_icp PRIVATE lidar_project_options)

include(CTest)
if(BUILD_TESTING)
  add_test(NAME bootstrap_help COMMAND lidar_icp --help)
  set_tests_properties(bootstrap_help PROPERTIES PASS_REGULAR_EXPRESSION "source to target")
  add_test(NAME bootstrap_version COMMAND lidar_icp --version)
  set_tests_properties(bootstrap_version PROPERTIES PASS_REGULAR_EXPRESSION "0.1.0")
  add_test(NAME bootstrap_rejects_unknown COMMAND lidar_icp --unknown)
  set_tests_properties(bootstrap_rejects_unknown PROPERTIES WILL_FAIL TRUE)
  if(LIDAR_ENABLE_PCL)
    add_test(NAME dependency_check COMMAND lidar_dependency_check)
    set_tests_properties(dependency_check PROPERTIES TIMEOUT 60)
  endif()
endif()

# Module owners add only their source/header/test files. The integrator owns
# these targets so separate worktrees can build without editing shared CMake.
if(LIDAR_ENABLE_PCL)
  foreach(module IN ITEMS cloud_processor pose_utils icp_registration)
    if(EXISTS "${PROJECT_SOURCE_DIR}/src/${module}.cpp")
      if(NOT EXISTS "${PROJECT_SOURCE_DIR}/include/${module}.h")
        message(FATAL_ERROR "${module}.cpp requires include/${module}.h")
      endif()
      if(module STREQUAL "cloud_processor")
        set(module_target cloud_processing)
      else()
        set(module_target ${module})
      endif()
      add_library(${module_target} STATIC "src/${module}.cpp")
      target_link_libraries(${module_target} PUBLIC lidar_project_options lidar_dependencies)
      if(module STREQUAL "icp_registration")
        if(NOT TARGET pose_utils)
          message(FATAL_ERROR "icp_registration requires src/pose_utils.cpp and its header")
        endif()
        target_link_libraries(icp_registration PUBLIC pose_utils)
      endif()
    endif()
  endforeach()

  if(EXISTS "${PROJECT_SOURCE_DIR}/tools/generate_test_data.cpp")
    add_executable(generate_test_data tools/generate_test_data.cpp)
    target_link_libraries(generate_test_data PRIVATE lidar_project_options lidar_dependencies)
  endif()

  if(BUILD_TESTING)
    foreach(test_name IN ITEMS test_cloud_processor test_pose test_icp)
      if(EXISTS "${PROJECT_SOURCE_DIR}/tests/${test_name}.cpp")
        if(test_name STREQUAL "test_cloud_processor")
          set(tested_target cloud_processing)
        elseif(test_name STREQUAL "test_pose")
          set(tested_target pose_utils)
        else()
          set(tested_target icp_registration)
        endif()
        if(NOT TARGET ${tested_target})
          message(FATAL_ERROR "${test_name} requires the ${tested_target} module")
        endif()
        add_executable(${test_name} "tests/${test_name}.cpp")
        target_link_libraries(${test_name} PRIVATE ${tested_target})
        add_test(NAME ${test_name} COMMAND ${test_name})
        set_tests_properties(${test_name} PROPERTIES TIMEOUT 120 WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
      endif()
    endforeach()
  endif()
endif()

if(TARGET cloud_processing AND TARGET icp_registration)
  add_library(registration_pipeline STATIC src/registration_pipeline.cpp)
  target_link_libraries(registration_pipeline PUBLIC cloud_processing icp_registration)
  target_link_libraries(lidar_icp PRIVATE registration_pipeline)
  target_compile_definitions(lidar_icp PRIVATE LIDAR_HAS_PIPELINE=1)
  if(BUILD_TESTING)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    add_test(NAME test_pipeline
      COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/test_pipeline.py"
        --exe $<TARGET_FILE:lidar_icp> --workdir "${PROJECT_BINARY_DIR}/test-pipeline")
    set_tests_properties(test_pipeline PROPERTIES TIMEOUT 120)
  endif()
endif()
